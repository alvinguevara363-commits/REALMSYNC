// --- RANKING ---
let servidores = [
  { id: 1, nombre: "Aetherion", juego: "Lineage II", cap: "x10", online: true, votos: 1450 },
  { id: 2, nombre: "Eclipse MU", juego: "MU Online", cap: "x100", online: true, votos: 1120 },
  { id: 3, nombre: "Rexall Online", juego: "Silkroad", cap: "Cap 110", online: true, votos: 980 }
];
const votosRealizados = new Set();

function renderizarRanking() {
  const contenedor = document.getElementById('lista-servidores');
  let ordenados = [...servidores].sort((a, b) => b.votos - a.votos);
  contenedor.innerHTML = ordenados.map((srv, index) => {
    const pos = index + 1;
    const claseTop = pos <= 3 ? top-${pos} : '';
    const yaVoto = votosRealizados.has(srv.id);
    return `
      <article class="server-item ${claseTop}">
        <div class="rank-num">#${pos}</div>
        <div class="server-info">
          <h3>${srv.nombre}</h3>
          <p><span class="status">${srv.online ? 'Online' : 'Offline'}</span> &nbsp;•&nbsp; ${srv.juego} &nbsp;•&nbsp; ${srv.cap}</p>
        </div>
        <div class="server-stats"><div><strong>${srv.votos}</strong> Votos</div></div>
        <button class="btn-vote" onclick="votar(${srv.id})" ${yaVoto ? 'disabled' : ''}>${yaVoto ? '✓ Votado' : '⭐ Votar'}</button>
      </article>
    `;
  }).join('');
}

function votar(id) {
  if (votosRealizados.has(id)) return;
  const srv = servidores.find(s => s.id === id);
  if (srv) { srv.votos++; votosRealizados.add(id); renderizarRanking(); }
}

// --- CHAT CUSTOM ---
function toggleEmojis() { document.getElementById('emoji-picker').classList.toggle('active'); }
function addEmoji(emoji) { const input = document.getElementById('realm-input'); input.value += emoji; input.focus(); toggleEmojis(); }

function enviarMensaje() {
  const input = document.getElementById('realm-input');
  const texto = input.value.trim();
  if (!texto) return;
  const chatBox = document.getElementById('realm-messages');
  const nuevoMsg = document.createElement('div');
  nuevoMsg.className = 'r-msg';
  const ahora = new Date();
  const hora = ahora.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});

  // Creación de nodos (Anti-XSS)
  const avatar = document.createElement('div'); avatar.className = 'r-avatar'; avatar.style.background = 'var(--cyan)'; avatar.style.color = '#000'; avatar.textContent = 'Tú';
  const content = document.createElement('div'); content.className = 'r-content';
  const header = document.createElement('div'); header.className = 'r-header';
  const author = document.createElement('span'); author.className = 'r-author'; author.textContent = 'Tú';
  const time = document.createElement('span'); time.className = 'r-time'; time.textContent = hora;
  const textDiv = document.createElement('div'); textDiv.className = 'r-text'; textDiv.textContent = texto;

  header.appendChild(author); header.appendChild(time);
  content.appendChild(header); content.appendChild(textDiv);
  nuevoMsg.appendChild(avatar); nuevoMsg.appendChild(content);
  
  chatBox.appendChild(nuevoMsg);
  input.value = '';
  chatBox.scrollTop = chatBox.scrollHeight;
}

function traducir(boton, textoTraducido) {
  const contenedor = boton.parentElement.querySelector('.r-text');
  contenedor.textContent = textoTraducido;
  contenedor.style.color = 'var(--cyan)';
  contenedor.style.fontStyle = 'italic';
  boton.style.display = 'none';
}

// --- SISTEMA DE LOGIN SECRETO Y ADMIN ---
function abrirLogin() {
  document.getElementById('login-overlay').classList.add('active');
  document.getElementById('admin-password').focus();
  document.getElementById('login-error').style.display = 'none';
}

function cerrarLogin() {
  document.getElementById('login-overlay').classList.remove('active');
  document.getElementById('admin-password').value = '';
}

function verificarLogin() {
  const pass = document.getElementById('admin-password').value;
  if (pass === 'admin') { // En producción, esto se valida en el servidor
    cerrarLogin();
    // Ocultar pestañas públicas y mostrar el panel de admin
    document.querySelectorAll('.view-section').forEach(sec => sec.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
    document.getElementById('admin-panel').classList.add('active');
    
    // Opcional: Podrías añadir dinámicamente un botón "Salir de Admin" aquí
  } else {
    document.getElementById('login-error').style.display = 'block';
    document.getElementById('admin-password').value = '';
  }
}

function banear(boton) {
  const fila = boton.closest('tr');
  const estado = fila.querySelector('td:nth-child(3)');
  estado.textContent = 'Baneado';
  estado.style.color = 'var(--danger)';
  boton.textContent = 'Desbanear';
  boton.classList.remove('btn-ban');
  boton.classList.add('btn-verify');
  boton.style.color = 'var(--success)';
}
